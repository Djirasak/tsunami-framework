ตัวกรอง Filters

 
<button type="button" class="btn" commandfor="drw-filter" command="show-modal"><i class="icon icon-settings" aria-hidden="true"></i>ตัวกรอง Filters</button>
<dialog class="drawer" id="drw-filter" aria-labelledby="drw-filter-title" closedby="any">
  <header>
    <h2 id="drw-filter-title">ตัวกรอง Filters</h2>
    <button type="button" class="btn" data-variant="ghost" data-size="icon" commandfor="drw-filter" command="close" aria-label="ปิด Close"><i class="icon icon-x" aria-hidden="true"></i></button>
  </header>
  <div class="stack">
    <div class="field">
      <label for="drw-status">สถานะ Status</label>
      <select id="drw-status">
        <option>ทั้งหมด All</option>
        <option>ชำระแล้ว Paid</option>
      </select>
    </div>
    <label class="field"><input type="checkbox"> เฉพาะคำสั่งซื้อใหม่ New orders only</label>
  </div>
  <footer>
    <button type="button" class="btn" commandfor="drw-filter" command="close">ยกเลิก Cancel</button>
    <button type="button" class="btn" data-variant="primary" commandfor="drw-filter" command="close">ใช้ตัวกรอง Apply</button>
  </footer>
</dialog>
