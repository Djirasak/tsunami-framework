ขั้นตอนของงาน Steps of a job

.icon หรือ .loader ที่อยู่แรกสุดใน event จะแทนที่จุด ใส่ aria-busy="true" ที่ขั้นที่กำลังทำ และ data-variant="pending" ที่ขั้นที่ยังไม่เริ่ม (ข้อความสีเทา จุดกลวง)

  1. สร้างบัญชี Creating your account
  2. ตั้งค่าร้านค้า Initializing your store
  3. นำเข้าสินค้า Importing products 
<ol class="timeline">
  <li data-variant="ok">
    <i class="icon icon-circle-check" aria-hidden="true"></i>
    สร้างบัญชี Creating your account
  </li>
  <li aria-busy="true">
    <span class="loader" role="status" aria-label="กำลังทำ In progress"></span>
    ตั้งค่าร้านค้า Initializing your store
  </li>
  <li data-variant="pending">นำเข้าสินค้า Importing products</li>

ขั้นตอนที่สคริปต์เล่นให้ดู Steps played by a script

สคริปต์ของหน้าเปลี่ยน attribute เอง: ขั้นที่กำลังทำได้ .loader พอเสร็จเปลี่ยนเป็นไอคอนเช็ก แล้วเริ่มขั้นถัดไป timeline ไม่มี JS ของตัวเอง

  1. สร้างบัญชี Creating your account
  2. ตั้งค่าร้านค้า Initializing your store
  3. นำเข้าสินค้า Importing products 
<div class="stack" data-gap="sm">
  <ol class="timeline" id="tl-steps">
    <li data-variant="pending">สร้างบัญชี Creating your account</li>
    <li data-variant="pending">ตั้งค่าร้านค้า Initializing your store</li>
    <li data-variant="pending">นำเข้าสินค้า Importing products</li>
  </ol>
  <div>
    <button type="button" class="btn" data-size="sm" id="tl-steps-play">เล่น Play</button>
  </div>
</div>
<script type="module">
  const steps = [...document.querySelectorAll("#tl-steps > li")];
  const play = document.getElementById("tl-steps-play");
  const spinner = '<span class="loader" role="status" aria-label="กำลังทำ In progress"></span>';
  const check = '<i class="icon icon-circle-check" aria-hidden="true"></i>';
  const start = (step) => {
    step.removeAttribute("data-variant");
    step.setAttribute("aria-busy", "true");
    step.insertAdjacentHTML("afterbegin", spinner);
  };
  play.addEventListener("click", () => {
    play.disabled = true;
    for (const step of steps) {
      step.querySelector(".icon, .loader")?.remove();
      step.removeAttribute("aria-busy");
      step.dataset.variant = "pending";
    }
    let current = 0;
    start(steps[current]);
    const timer = setInterval(() => {
      const step = steps[current];
      step.removeAttribute("aria-busy");
      step.dataset.variant = "ok";
      step.querySelector(".loader").outerHTML = check;
      current += 1;
      if (steps[current]) return start(steps[current]);
      clearInterval(timer);
      play.disabled = false;
    }, 1500);
  });
</script>

ฟอร์มเพิ่มบันทึก A form for adding a note

event แรกเป็น <form> ที่มี <textarea> และ <footer> ปุ่มกับคำอธิบาย <small> ในตัวอย่าง สคริปต์ของหน้าเพิ่มบันทึกใหม่ลงใต้ฟอร์มเมื่อกดบันทึก ส่วนปุ่มยกเลิกเป็น type="reset"

  1. แก้ไขบันทึกได้ภายใน 5 นาทีหลังบันทึก Notes are editable for 5 minutes.
  2. ส่งอีเมลยืนยันให้ลูกค้าแล้ว Confirmation email sent 3 ต.ค. 2026 00:36
  3. สถานะจัดส่งเปลี่ยนจาก Unfulfilled เป็น On hold 3 ต.ค. 2026 00:35 
<ol class="timeline" id="tl-notes">
  <li>
    <label for="tl-note">เพิ่มบันทึก Add a note</label>
    <form id="tl-note-form">
      <textarea id="tl-note" name="note" placeholder="เขียนบันทึกคำสั่งซื้อ Add order note..." required></textarea>
      <footer>
        <button type="submit" class="btn" data-variant="primary" data-size="sm">บันทึก Save</button>
        <button type="reset" class="btn" data-size="sm">ยกเลิก Cancel</button>
        <small>แก้ไขบันทึกได้ภายใน 5 นาทีหลังบันทึก Notes are editable for 5 minutes.</small>
      </footer>
    </form>
  </li>
  <li>
    ส่งอีเมลยืนยันให้ลูกค้าแล้ว Confirmation email sent
    <time datetime="2026-10-03T00:36">3 ต.ค. 2026 00:36</time>
  </li>
  <li>
    สถานะจัดส่งเปลี่ยนจาก Unfulfilled เป็น On hold
    <time datetime="2026-10-03T00:35">3 ต.ค. 2026 00:35</time>
  </li>
</ol>
<script type="module">
  const form = document.getElementById("tl-note-form");
  form.addEventListener("submit", (event) => {
    event.preventDefault();
    const now = new Date();
    const li = document.createElement("li");
    const article = document.createElement("article");
    const name = document.createElement("h4");
    const time = document.createElement("time");
    const text = document.createElement("p");
    name.textContent = "คุณ You";
    time.dateTime = now.toISOString();
    time.textContent = now.toLocaleString("th-TH", { dateStyle: "medium", timeStyle: "short" });
    text.textContent = form.elements.note.value;
    article.append(name, time, text);
    li.append(article);
    form.closest("li").after(li);
    form.reset();
  });
</script>

Attributes

Attribute / element Values Description
ol.timeline รายการเหตุการณ์ เรียงใหม่สุดไว้บน
> li หนึ่งเหตุการณ์ มีจุดและเส้นต่อไปยังเหตุการณ์ถัดไป (ยกเว้นอันสุดท้าย)
li[data-variant] ok, danger, pending; ไม่ใส่ = ปกติ (จุดสีเทา) ok จุดและไอคอนแรกสีเขียว, danger จุดและข้อความสีแดง, pending ขั้นที่ยังไม่เริ่ม ข้อความสีเทา จุดกลวง
li[aria-busy] true บอก screen reader ว่าขั้นนี้กำลังทำ (คู่กับ .loader)
li > .icon:first-child, li > .loader:first-child ไอคอนหรือ loader ที่อยู่แรกสุดแทนที่จุด
time datetime เวลาของเหตุการณ์ ขึ้นบรรทัดใหม่ ตัวเล็กสีเทา
details > summary open เหตุการณ์ที่เปิดดูรายละเอียดได้ summary มีลูกศรให้เอง
dl > dt + dd รายละเอียดแบบชื่อ/ค่า สองคอลัมน์ในกรอบ
article บันทึก: h4 ผู้เขียน และ time อยู่บรรทัดเดียวกัน ตามด้วย p และ footer
footer แถวปุ่มของบันทึกหรือฟอร์ม small ในนั้นเป็นคำอธิบายสีเทา
form, textarea ฟอร์มเพิ่มบันทึกใน event; textarea เต็มความกว้าง
